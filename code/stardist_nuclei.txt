# -*- coding: utf-8 -*-
"""
Simple 3D Nuclei Segmentation using StarDist.

This script processes all 3D TIFF images in a specified folder, segments the
nuclei using a pre-trained StarDist model, and saves the results.
"""

import numpy as np
from pathlib import Path
from tifffile import imread, imwrite
from stardist.models import StarDist3D
from csbdeep.utils import normalize

# =============================================================================
# --- 1. CONFIGURATION ---
# Please edit the settings in this section.
# =============================================================================

# --- Path Settings ---
# Set the path to the folder containing your 3D TIFF images.
# Use 'r' before the string on Windows to handle backslashes correctly.
INPUT_DIR = Path(r"C:\Users\patri\Desktop\MASTER\TFM\YDRAY-trainingDataset\trainingDataset\nuclei\0.1")

# Set the name for the output folder. It will be created inside the input folder.
OUTPUT_DIR_NAME = "output_segmented_nuclei"

# --- Model Parameters ---
# Probability threshold: Higher values detect fewer, more certain objects.
# Good values are typically between 0.5 and 0.7.
PROB_THRESHOLD = 0.64

# Non-maximum suppression (NMS) threshold: Higher values allow more object overlap.
# Good values are typically between 0.2 and 0.5.
NMS_THRESHOLD = 0.2

# =============================================================================
# --- End of Configuration ---
# You shouldn't need to edit anything below this line.
# =============================================================================


# --- 2. SETUP ---
# Create the full path for the output directory and make sure it exists.
output_dir = INPUT_DIR / OUTPUT_DIR_NAME
output_dir.mkdir(parents=True, exist_ok=True)

print(f"Input folder: {INPUT_DIR}")
print(f"Output will be saved to: {output_dir}")
print("-" * 30)

# Load the pre-trained StarDist model for 3D nuclei.
# '3D_versatile_fluo' is a robust model for general fluorescence images.
print("Loading StarDist 3D model...")
model = StarDist3D.from_pretrained("3D_versatile_fluo")
print("Model loaded successfully.")
print("-" * 30)


# --- 3. PROCESSING LOOP ---
# Find all files ending with .tif or .tiff in the input folder.
image_files = list(INPUT_DIR.glob('*.tif*'))

# Check if any images were found before starting.
if not image_files:
    print(f" Error: No TIFF files were found in '{INPUT_DIR}'")
else:
    print(f"Found {len(image_files)} images to process...")
    
    # Loop over each found image file.
    for image_path in image_files:
        print(f"-> Processing: {image_path.name}")

        # Load the 3D image from the file.
        img = imread(image_path)

        # Normalize the image. This is a crucial step for deep learning models.
        # It scales the pixel intensity values to a range the model expects.
        img_norm = normalize(img, 1, 99.8, axis=(0, 1, 2))

        # Run the StarDist prediction to find and label each nucleus.
        labels, _ = model.predict_instances(
            img_norm,
            prob_thresh=PROB_THRESHOLD,
            nms_thresh=NMS_THRESHOLD
        )

        # Create a new filename for the output mask.
        output_filename = f"{image_path.stem}_segmented.tif"
        output_path = output_dir / output_filename
        
        # Save the resulting label mask as a 32-bit TIFF image.
        imwrite(output_path, labels.astype(np.uint32), bigtiff=True)

    print("-" * 30)
    print("Segmentation complete!")
    print(f"All results have been saved in: {output_dir}")
